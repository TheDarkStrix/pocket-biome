var input_actions := 0
func _init() -> void: call_deferred("run")
func fail(text: String) -> void:
 printerr("OPENING TOUCH FAIL: "+text); quit(1)
func button(text: String) -> Button:
 var parent: Node = h.launch_screen if h.launch_menu else h.modal_box
 for action in parent.find_children("*","Button",true,false):
  if action.text == text: return action
 return null
func touch(at: Vector2, down: bool) -> void:
 var event := InputEventScreenTouch.new(); event.index=0; event.position=at; event.pressed=down
 Input.parse_input_event(event);Input.flush_buffered_events()
func click_at(at: Vector2) -> void:
 touch(at,true);await process_frame;touch(at,false);await process_frame
 input_actions+=1
func click(action: Control) -> void:
 if action == null: fail("missing requested control");return
 if h.modal.visible and h.modal_scroll.is_ancestor_of(action):h.modal_scroll.ensure_control_visible(action)
 for frame in range(4):await process_frame
 await click_at(action.get_global_rect().get_center())
func craft_action() -> Button:
 for action in h.modal_box.find_children("*","Button",true,false):
  if action.get_meta("build_confirm","")=="workbench":return action
 return null
