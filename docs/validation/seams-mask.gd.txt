extends SceneTree
class Tiles extends Node2D:
 var half:=58.0
 var turn:=0
 var harvested:=false
 var cells={Vector2i(0,0):"oak",Vector2i(0,1):"fiber",Vector2i(1,0):"stone",Vector2i(1,1):"grass",Vector2i(-1,0):"water",Vector2i(2,1):"water"}
 func rotated(c: Vector2) -> Vector2:
  for i in range(turn):c=Vector2(-c.y,c.x)
  return c
 func project(c: Vector2) -> Vector2:
  var p:=rotated(c);return Vector2((p.x-p.y)*half,(p.x+p.y)*half/2)
 func screen(c: Vector2) -> Vector2:return Vector2(400,280)+project(c)*1.6
 func _draw() -> void:
  var tiles: Dictionary={}
  for cell in cells:tiles["%d,%d" % [cell.x,cell.y]]={"type":cells[cell],"size":1}
  if half==58:preload("res://scripts/PaintedJoins.gd").draw(self,tiles,screen,1.6)
  var keys:=cells.keys()
  keys.sort_custom(func(a,b):
   var ca:=rotated(Vector2(a));var cb:=rotated(Vector2(b))
   return ca.x<cb.x if is_equal_approx(ca.x+ca.y,cb.x+cb.y) else ca.x+ca.y<cb.x+cb.y)
  for key in keys:
   var mask: int=preload("res://scripts/PaintedJoins.gd").front_mask(tiles,"%d,%d" % [key.x,key.y],turn) if half==58 else 0
   preload("res://scripts/TileArtwork.gd").draw_tile(self,cells[key],screen(key),132.0/418.0*1.6,Color.WHITE,"",harvested,0,mask)
func _init() -> void:call_deferred("run")
func run() -> void:
 var viewport:=SubViewport.new();viewport.size=Vector2i(800,640);viewport.transparent_bg=true;viewport.world_2d=World2D.new();viewport.render_target_update_mode=SubViewport.UPDATE_ALWAYS;root.add_child(viewport)
 var painter:=Tiles.new();viewport.add_child(painter)
 var reports:=[]
 for half in [66.0,58.0]:
  for harvested in [false,true]:
   for turn in range(4):
    painter.half=half;painter.harvested=harvested;painter.turn=turn;painter.queue_redraw()
    for frame in range(3):await process_frame
    await RenderingServer.frame_post_draw
    var image: Image=viewport.get_texture().get_image()

    for cell in painter.cells:
     for direction in [Vector2i.RIGHT,Vector2i.DOWN]:
      if not painter.cells.has(cell+direction):continue
      var a:=Vector2(cell)+Vector2(0.5,-0.5) if direction==Vector2i.RIGHT else Vector2(cell)+Vector2(-0.5,0.5)
      var b:=Vector2(cell)+Vector2(0.5,0.5)
      for sample in range(3,38):
       var point:=painter.screen(a.lerp(b,sample/40.0))+Vector2(0,-7)*1.6
       var alpha: float=image.get_pixelv(Vector2i(point)).a
       minimum=minf(minimum,alpha);count+=1
       if alpha<0.7:gaps+=1
    reports.append({"half_width":half,"harvested":harvested,"turn":turn,"samples":count,"gaps":gaps,"min_alpha":minimum})
    if turn==0:image.save_png("/workspace/runtime/seam-review/mask-%d-%s.png" % [int(half),"harvested" if harvested else "ready"])
 var report: Dictionary={"source_sha256":JSON.parse_string(FileAccess.get_file_as_string("res://data/build_info.json")).source_sha256,"cases":reports,"scope":"Actual original painted textures on a transparent OpenGL viewport, 210 shared-edge samples per view/state/pitch. Alpha coverage is not aesthetic acceptance."}
 var f:=FileAccess.open("/workspace/runtime/seam-review/coverage.json",FileAccess.WRITE);f.store_string(JSON.stringify(report,"  "));f.close();print("SEAM COVERAGE: "+JSON.stringify(reports))
 viewport.queue_free();await process_frame;quit()
